    \section{Introduction}

    This report will discuss the design and methodologies of the SSME code, as well as the results of the code.

    \section{Oxygen}

    \subsection{First Enthalpy Exchanger}
    We begin with computing the equivalence ratio $\phi$  for the oxygen first enthalpy exchanger. 
    The equivalence ratio is defined as the ratio of the mass of fuel to the mass of oxidizer. 
    The mass of fuel is the mass of hydrogen, and the mass of oxidizer is the mass of oxygen. 
    The mass of hydrogen is $\phi \frac{mol_{O_{2}}}{mol_{H_{2}}} = \Phi$.


    \vspace*{6pt}

    From here we are to calculate the stndard heat of formation by:
    
    \begin{equation}
        \Delta \hat{h}_{H_{2}} = \hat{h}_{H_{2}}(850) + \hat{R}T^{2}\int^{pi}_{pf} \frac{1}{p} \frac{-\hat{b}_{H_{2}}p}{\hat{R}T^{2}} dp
    \end{equation}

    \vspace*{6pt}

    \begin{equation}
        \Delta \hat{h}_{O_{2}} = \hat{h}_{O_{2}}(530) + \hat{R}T^{2}\int^{pi}_{pf} \frac{1}{p} \frac{-\hat{b}_{O_{2}}p}{\hat{R}T^{2}} dp
    \end{equation}

    \begin{center}
        \begin{tabular}{|c|c|}
            \hline
            Species & $\hat{b}$ \\
            \hline
            $H_{2}$ & 0.0266 \\
            \hline
            $O_{2}$ & 0.0318 \\
            \hline
            $H_{2}O$ & 0.03049 \\
            \hline
            $N_{2}$ & 0.0391 \\
            \hline
            $F_{2}$ & 0.02896 \\
            \hline
            $HF$ & 0.0739 \\
            \hline
        \end{tabular}
    \end{center}
    
    Where $\hat{b}$ is read from a table developed in \verb|exchanger.py| file. $\Delta \hat{h}_{i}$ is calculated from both 
    hydrogen and oxygen. Once we calculate $\Delta \hat{h}_{i}$ for each, we can calculate the enthalpy of the mixture.

    \begin{equation}
        \Delta \hat{h}_{mix} = \Phi \Delta \hat{h}_{H_{2}} + \Delta \hat{h}_{O_{2}}
    \end{equation}
    
    With $\Phi = 2\phi$ moles of $H_{2}$ per mole of $O_{2}$, complete combustion (no dissociation) gives

    \begin{equation}
        O_{2} + \Phi H_{2} \rightarrow 2H_{2}O + (\Phi-2)H_{2}
    \end{equation}

    The energy available to heat the products is the propellant enthalpy plus the heat released forming two moles of water:

    \begin{equation}
        \Delta \hat{h}_{total} = \Phi \Delta \hat{h}_{H_{2}} + \Delta \hat{h}_{O_{2}} - 2\Delta \hat{h}^{298}_{f,H_{2}O}
    \end{equation}

    The first estimate of the chamber temperature, $T_{FEE}$, is the single temperature at which the products absorb exactly
    that energy. It is found with \verb|scipy.optimize.brentq|:

    \begin{equation}
        2\left[\hat{h}_{H_{2}O}(T_{FEE}) - \hat{h}_{H_{2}O}(298)\right] + (\Phi-2)\left[\hat{h}_{H_{2}}(T_{FEE}) - \hat{h}_{H_{2}}(298)\right] = \Delta \hat{h}_{total}
    \end{equation}

    This gives $T_{FEE} \approx 4617$ K. Because it ignores dissociation it is an upper bound; the chamber temperature
    comes from the second enthalpy exchanger in Section \ref{sec:chamber-temperature}.

    \vspace*{6pt}

    \subsection{Combustion Chamber}
    For the Combustion Chamber, we have the two equations below:

    \begin{equation}
        H_{2}O \rightarrow OH + H 
    \end{equation}

    \begin{equation}
        H_{2} \rightarrow 2H
    \end{equation}

    The Gibbs Free Energy for each species is given by:
    
    \begin{equation}
        \hat{g}_{H_{2}O} = \verb|h2o_table_g|(T_{C})
    \end{equation}

    \begin{equation}
        \hat{g}_{H_{2}} = \verb|h2_table_g|(T_{C})
    \end{equation}

    \begin{equation}
        \hat{g}_{OH} = \verb|oh_table_g|(T_{C})
    \end{equation}

    \begin{equation}
        \hat{g}_{H} = \verb|h_table_g|(T_{C})
    \end{equation}

    \vspace*{6pt}

    We can then calculate the Gibbs Free Energy of the reaction by:

    \begin{equation}
        \Delta \hat{g}_{H_{2}} = 2\hat{g}_{H} - \hat{g}_{H_{2}}
    \end{equation}

    \begin{equation}
        \Delta \hat{g}_{H_{2}O} = \hat{g}_{H} + \hat{g}_{OH} - \hat{g}_{H_{2}O}
    \end{equation}

    Now is the point where we involve the \textbf{Law of Mass Action}:

    \begin{center}
        \begin{tabular}{|c|c|c|c|}
            \hline
            Species & $\nu_{(i)} '$ & $\nu_{(i)} ''$ & $\nu_{(i)} '' - \nu_{(i)} '$ \\
            \hline
            $H_{2}O$ & 1 & 0 & -1 \\
            \hline
            $OH$ & 0 & 1 & 1 \\
            \hline
            $H$ & 0 & 1 & 1 \\
            \hline
        \end{tabular}
    \end{center}

    We an now use:

    \begin{equation}
        K_{n} = \prod^{N_{s}}_{i=1} X_{(n)}^{(\nu_{(i)} '' - \nu_{(i)} ')} = (\frac{p}{p_{a}})^{-\sigma_{v}}K_{p}
    \end{equation}

    Where;

    \begin{equation}
        \sigma_{v} = \sum^{N_{s}}_{i=1} (\nu_{(i)} '' - \nu_{(i)} ')
    \end{equation}

    \begin{equation}
        K_{p} = e^{- \frac{\Delta G}{\hat{R}T}}
    \end{equation}

    \begin{equation}
        \Delta G = \sum^{N_{s}}_{i=1} (\nu_{(i)} '' - \nu_{(i)} ') \hat{g}_{(i)}
    \end{equation}

    Both reactions have $\sigma_{v} = +1$, so we can solve for the equilibrium constant $K_{n}$.

    \begin{equation}
        K_{H_{2}O} = \left(\frac{204 atm}{1 atm}\right)^{-1} e^{-\frac{\Delta \hat{g}_{H_{2}O}}{\hat{R}T_{C}}}
    \end{equation}

    \begin{equation}
        K_{H_{2}} = \left(\frac{204 atm}{1 atm}\right)^{-1} e^{-\frac{\Delta \hat{g}_{H_{2}}}{\hat{R}T_{C}}}
    \end{equation}

    Now we move into the chemistry problem. 

    \textbf{Atom Balance}:

    \begin{equation}
        H_{2} + l O_{2} \rightarrow N_{H_{2}O} H_{2}O + N_{H_{2}} H_{2} + N_{H}H + N_{OH}OH
    \end{equation}

    Recall, due to the fule being Hydrogen;

    \begin{equation}
        mass_{fuel} = 2.02 
    \end{equation}

    \begin{equation}
        mass_{Oxidizer} =  \frac{mass_{fuel}}{\phi}
    \end{equation}

    \begin{equation}
        l = \frac{mass_{Oxidizer}}{2 \hat{m}_{O_{2}}}
    \end{equation}

    Where $\hat{m}_{O_{2}}$ is the molar mass of Oxygen. And $\phi$ is the Fuel to Oxidizer ratio, and not the equivalence ratio.
    Revisiting the \textbf{Atom Balance} equation, we can now solve for the number of moles of each species. 

    The atom balances for hydrogen and oxygen are:

    \begin{equation}
        H: \quad 2 = 2N_{H_{2}O} + 2N_{H_{2}} + N_{H} + N_{OH}
    \end{equation}

    \begin{equation}
        O: \quad 2l = N_{H_{2}O} + N_{OH}
    \end{equation}

    That is four unknowns with only two equations, so the Law of Mass Action supplies the other two.
    The mole fractions are normalized by the full total,
    \begin{equation}
        X_{(i)} = \frac{N_{(i)}}{N_{Total}}, \qquad N_{Total} = N_{H_{2}O} + N_{H_{2}} + N_{H} + N_{OH},
    \end{equation}
    so the two equilibrium relations become
    \begin{equation}
        K_{H_{2}O} = \frac{X_{H} X_{OH}}{X_{H_{2}O}} = \frac{N_{H} N_{OH}}{N_{H_{2}O} N_{Total}}
    \end{equation}
    \begin{equation}
        K_{H_{2}} = \frac{X_{H}^{2}}{X_{H_{2}}} = \frac{N_{H}^{2}}{N_{H_{2}} N_{Total}}
    \end{equation}

    The four coupled equations (two atom balances and two equilibrium relations) are solved with
    \verb|scipy.optimize.fsolve| in terms of $\ln N_{(i)}$, which keeps every mole number positive.
    The starting guess is complete combustion, $N_{H_{2}O} = 2l$ and $N_{H_{2}} = 1 - 2l$.
    At the chamber temperature $T_{c}$ (Section \ref{sec:chamber-temperature}) the solution is:

    \begin{equation}
        N_{H_{2}O} = 0.6763, \quad N_{H_{2}} = 0.2559, \quad N_{H} = 0.0530, \quad N_{OH} = 0.0827
    \end{equation}

    \begin{center}
        \begin{tabular}{|c|c|c|c|c|c|c|c|}
            \hline
            Species & $N_{(i)}$ & $X_{(i)}$ & $m_{(i)}$ &  $Y_{(i)}$ & $\hat{c}_{p(i)}$ & $c_{p(i)}$ & $Y_{(i)}c_{p}$ \\
            \hline
            $H_{2}O$ & $N_{H_{2}O}$ & $X_{H_{2}O} = \frac{N_{H_{2}O}}{N_{Total}}$ & $\hat{m}N_{H_{2}O}$ & $\frac{\hat{m}}{m_{(i)}}$ & $\hat{c}_{p(H_{2}O)}(T_{C})$ & $\frac{\hat{c}_{p}}{\hat{m}}$ & $Y_{H_{2}O}c_{p}$ \\
            \hline
            $H_{2}$ & $N_{H_{2}}$ & $X_{H_{2}} = \frac{N_{H_{2}}}{N_{Total}}$ & $\hat{m}N_{H_{2}}$ & $\frac{\hat{m}}{m_{(i)}}$ & $\hat{c}_{p(H_{2})}(T_{C})$ & $\frac{\hat{c}_{p}}{\hat{m}}$ & $Y_{H_{2}}c_{p}$ \\
            \hline
            $H$ & $N_{H}$ & $X_{H} = \frac{N_{H}}{N_{Total}}$ & $\hat{m}N_{H}$ & $\frac{\hat{m}}{m_{(i)}}$ & $\hat{c}_{p(H)}(T_{C})$ & $\frac{\hat{c}_{p}}{\hat{m}}$ & $Y_{H}c_{p}$ \\
            \hline
            $OH$ & $N_{OH}$ & $X_{OH} = \frac{N_{OH}}{N_{Total}}$ & $\hat{m}N_{OH}$ & $\frac{\hat{m}}{m_{(i)}}$ & $\hat{c}_{p(OH)}(T_{C})$ & $\frac{\hat{c}_{p}}{\hat{m}}$ & $Y_{OH}c_{p}$ \\
            \hline
        \end{tabular}
    \end{center}

    Now we can calculate $c_{p}$, we sum the $Y_{(i)}c_{p}$ values for each species.

    \begin{equation}
        c_{p} = \sum_{i} Y_{(i)}c_{p}
    \end{equation}

    \begin{equation}
        c_{p} = Y_{H_{2}O}c_{p} + Y_{H_{2}}c_{p} + Y_{H}c_{p} + Y_{OH}c_{p}
    \end{equation}

    We perform now calulate the gas constant $R$.

    \begin{equation}
        R = \frac{\hat{R}}{m_{Total}}
    \end{equation}
    where $m_{Total} = \sum_{i} X_{(i)} m_{(i)}$ is the mixture molar mass. The mole fractions are normalized by
    $N_{Total} = N_{H_{2}O} + N_{H_{2}} + N_{H} + N_{OH}$, which exceeds one once dissociation produces H and OH.

    \begin{equation}
        c_{v} = c_{p} - R
    \end{equation}

    \begin{equation}
        \gamma = \frac{c_{p}}{c_{p} - \frac{\hat{R}}{\sum_{i} X_{(i)}m_{(i)}}} = \frac{c_{p}}{c_{p}- \frac{\hat{R}}{m_{Total}}} = \frac{c_{p}}{c_{p}- R} = \frac{c_{p}}{c_{v}}
    \end{equation}
    
    \subsection{Chamber Temperature (Second Enthalpy Exchanger)}
    \label{sec:chamber-temperature}

    Breaking $H_{2}O$ and $H_{2}$ apart into $H$ and $OH$ absorbs energy, so the dissociated products reach a lower
    temperature than $T_{FEE}$. The composition depends on temperature through $K_{n}$, and the temperature depends on the
    composition through the energy balance, so the two are solved together. For a trial temperature the combustion-chamber
    step gives the equilibrium mixture, and the energy balance per mole of $O_{2}$ is checked:

    \begin{equation}
        \sum_{i} N_{(i)} \left[\Delta \hat{h}^{298}_{f(i)} + \hat{h}_{(i)}(T_{c}) - \hat{h}_{(i)}(298)\right] = \Phi \Delta \hat{h}_{H_{2}} + \Delta \hat{h}_{O_{2}}
    \end{equation}

    \verb|brentq| adjusts the temperature between 2000 K and $T_{FEE}$ until both sides match. The sensible enthalpies of
    $H_{2}$, $O_{2}$ and $H_{2}O$ come straight from the tables, $H$ uses its constant $c_{p}$, and $OH$ is built from its
    tabulated enthalpy of formation:

    \begin{equation}
        \hat{h}_{OH}(T) - \hat{h}_{OH}(298) = \Delta \hat{h}_{f,OH}(T) - \Delta \hat{h}_{f,OH}(298) + \frac{1}{2}\left[\hat{h}_{H_{2}}(T) - \hat{h}_{H_{2}}(298)\right] + \frac{1}{2}\left[\hat{h}_{O_{2}}(T) - \hat{h}_{O_{2}}(298)\right]
    \end{equation}

    The same construction applied to $H_{2}O$ reproduces its tabulated sensible enthalpy within 0.07\%. The result is
    $T_{c} \approx 3949$ K, which is used for $\gamma$, $R$ and the nozzle. A Cantera equilibrium calculation with the same
    inputs gives 3881 K; most of the difference is that this model leaves out $O$ and $O_{2}$.

    \subsection{Nozzle Performance}
    We begin by defining the Area Ratio:

    \begin{equation}
        A_{r} = \frac{\pi R_{e}^{2}}{\pi R_{t}^{2}}
    \end{equation}

    Allowing for us to solve for the Mach number:

    \begin{equation}
        A_{r} = \left(\frac{\gamma + 1}{2}\right)^{-\frac{\gamma + 1}{2(\gamma - 1)}} \frac{1}{M_{e}} \left[1 + \frac{\gamma - 1}{2} M_{e}^{2}\right]^{\frac{\gamma + 1}{2(\gamma - 1)}}
    \end{equation}

    We can solve for Mach number $M_{e}$ by making use of the \verb|scipy.optimize.fsolve| function, which returns the roots 
    of a non-linear function. After solving for $M_{e}$ we can calculate the exit velocity $v_{e}$, exit Pressure $p_{e}$ and exit Temperature $T_{e}$.

    \begin{equation}
        \label{eq:exit-temperature}
        T_{e} = T_{c} \left[1 + \frac{\gamma - 1}{2} M_{e}^{2}\right]^{-1} 
    \end{equation}

    \begin{equation}
        \label{eq:exit-pressure}
        p_{e} = p_{c} \left[1 + \frac{\gamma - 1}{2} M_{e}^{2}\right]^{-\frac{\gamma}{\gamma - 1}}
    \end{equation}

    \begin{equation}
        \label{eq:exit-velocity}
        v_{e} = M_{e} \sqrt{\gamma R T_{e}}
    \end{equation}

    After solving for (Eq. \ref{eq:exit-velocity}), (Eq. \ref{eq:exit-pressure}) and (Eq. \ref{eq:exit-temperature}) we can calculate the mass flow rate $\dot{m}$.

    \begin{equation}
        \label{eq:mass-flow-rate}
        \dot{m} = \frac{A_{t} p_{c} \sqrt{\gamma}}{\sqrt{R T_{c}}} \left[ \frac{2}{\gamma + 1} \right]^{\frac{\gamma + 1}{2(\gamma - 1)}}
    \end{equation}

    Then once you have the mass flow rate you can calculate the thrust $\mathcal{T}$, specific impulse $I_{sp}$ and the thrust coefficient $C_{T}$.

    \begin{equation}
        \label{eq:thrust}
        \mathcal{T} = \dot{m} v_{e} + (p_{e} - p_{a}) A_{e}
    \end{equation}
    where $p_{a}$ is the ambient pressure ($p_{a} = 0$ in vacuum, $p_{a} = 101325$ Pa at sea level).

    \begin{equation}
        \label{eq:specific-impulse}
        I_{sp} = \frac{\mathcal{T}}{\dot{m} g_{0}}
    \end{equation}

    \begin{equation}
        \label{eq:thrust-coefficient}
        C_{T} = \frac{\mathcal{T}}{p_{c} A_{t}}
    \end{equation}
